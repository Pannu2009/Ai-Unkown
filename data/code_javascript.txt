Q: How do you throttle?
A: At most once per window.
function throttle(fn, ms) {
  let last = 0;
  return (...a) => {
    const now = Date.now();
    if (now - last > ms) {
      last = now;
      fn(...a);
    }
  };
}

Q: How do you memoize?
A: Cache by arguments.
function memo(fn) {

  return (x) => {

    return cache.get(x);
  };
}

Q: How do you shuffle?

function shuffle(a) {
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

Q: How do you do matrix math in JS?
A: Loops over typed arrays. That is what brain.js does.
function matvec(w, x, rows, cols) {
  const out = new Float32Array(rows);
  for (let r = 0; r < rows; r++) {
    let s = 0;
    for (let c = 0; c < cols; c++) s += w[r * cols + c] * x[c];
    out[r] = s;
  }
  return out;
}

Q: How do you do softmax in JS?
A: Same math as Python.
function softmax(x) {
  const m = Math.max(...x);
  const e = x.map(v => Math.exp(v - m));
  const s = e.reduce((a, b) => a + b, 0);
  return e.map(v => v / s);
}

Q: How do you do layer norm in JS?
A: Per-vector normalize.
function layernorm(x, g, b) {
  const m = x.reduce((a, v) => a + v, 0) / x.length;
  const v = x.reduce((a, vv) => a + (vv - m) ** 2, 0) / x.length;
  return x.map((vv, i) => g[i] * (vv - m) / Math.sqrt(v + 1e-5) + b[i]);
}

Q: How do you sample from probabilities in JS?
A: Cumulative random.
function sample(probs, temp = 1) {
  const scaled = probs.map(p => Math.pow(p, 1 / temp));
  const s = scaled.reduce((a, b) => a + b, 0);
  let r = Math.random() * s;
  for (let i = 0; i < scaled.length; i++) {
    r -= scaled[i];
    if (r <= 0) return i;
  }
  return scaled.length - 1;
}
