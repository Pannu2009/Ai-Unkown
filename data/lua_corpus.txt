-- magnitude check
local function nearestPlayer(pos)
    local best, bestDist
    for _, plr in ipairs(Players:GetPlayers()) do
        local char = plr.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if hrp then
            local d = (hrp.Position - pos).Magnitude
            if not bestDist or d < bestDist then
                best, bestDist = plr, d
            end
        end
    end
    return best, bestDist
end

-- camera shake: client
local cam = workspace.CurrentCamera
local function shake(intensity, time)
    local t = 0
    while t < time do
        t += task.wait()
        cam.CFrame = cam.CFrame * CFrame.new(
            math.random(-intensity, intensity),
            math.random(-intensity, intensity), 0)
    end
end

-- spring math
local function spring(pos, target, vel, stiff, damp, dt)
    local force = (target - pos) * stiff
    vel = (vel + force * dt) * (1 - damp * dt)
    return pos + vel * dt, vel
end

-- weighted random
local loot = { {n="common", w=70}, {n="rare", w=25}, {n="epic", w=5} }
local function roll()
    local total = 0
    for _, e in ipairs(loot) do total += e.w end
    local r = math.random() * total
    for _, e in ipairs(loot) do
        r -= e.w
        if r <= 0 then return e.n end
    end
end

-- cooldown helper
local Cooldown = {}
Cooldown.__index = Cooldown
function Cooldown.new(dur)
    return setmetatable({ dur = dur, last = 0 }, Cooldown)
end
function Cooldown:ready()
    return os.clock() - self.last >= self.dur
end
function Cooldown:use()
    self.last = os.clock()
end

-- signal / event bus
local Signal = {}
Signal.__index = Signal
function Signal.new()
    return setmetatable({ _c = {} }, Signal)
end
function Signal:connect(fn)
    table.insert(self._c, fn)
    return function()
        for i, f in ipairs(self._c) do
            if f == fn then table.remove(self._c, i) break end
        end
    end
end
function Signal:fire(...)
    for _, fn in ipairs(self._c) do task.spawn(fn, ...) end
end

-- state machine
local State = { Idle = 1, Chase = 2, Attack = 3 }
local mob = { state = State.Idle }
function mob:update(dt, target)
    if self.state == State.Idle then
        if (target.Position - self.pos).Magnitude < 20 then
            self.state = State.Chase
        end
    elseif self.state == State.Chase then
        self:moveToward(target.Position, dt)
        if (target.Position - self.pos).Magnitude < 5 then
            self.state = State.Attack
        end
    elseif self.state == State.Attack then
        self:attack(target)
        self.state = State.Chase
    end
end

-- binary search
local function bsearch(arr, x)
    local lo, hi = 1, #arr
    while lo <= hi do
        local mid = math.floor((lo + hi) / 2)
        if arr[mid] == x then return mid
        elseif arr[mid] < x then lo = mid + 1
        else hi = mid - 1 end
    end
end

-- shuffle
local function shuffle(t)
    for i = #t, 2, -1 do
        local j = math.random(i)
        t[i], t[j] = t[j], t[i]
    end
end

-- deep copy
local function deepcopy(o)
    if type(o) ~= "table" then return o end
    local c = {}
    for k, v in pairs(o) do c[deepcopy(k)] = deepcopy(v) end
    return setmetatable(c, getmetatable(o))
end

-- datastore session lock pattern
local sessionStore = DataStoreService:GetDataStore("Sessions")
local function tryLock(key, id)
    return sessionStore:UpdateAsync(key, function(old)
        if old and old.lock and os.time() - old.lock < 300 then return nil end
        return { lock = os.time(), owner = id }
    end)
end

-- rate limit helper
local buckets = {}
local function allowed(key, max, window)
    local now = os.clock()
    local b = buckets[key] or {}
    local fresh = {}
    for _, t in ipairs(b) do
        if now - t < window then table.insert(fresh, t) end
    end
    if #fresh >= max then return false end
    table.insert(fresh, now)
    buckets[key] = fresh
    return true
end
