Q: How do you flatten a dict?
A: Recursion with dotted keys.
def flat(d, pre=""):
    out = {}
    for k, v in d.items():
        if isinstance(v, dict):
            out.update(flat(v, pre + k + "."))
        else:
            out[pre + k] = v
    return out

Q: How do you unflatten?
A: Split dotted keys.
def unflat(d):
    out = {}
    for k, v in d.items():
        cur = out
        *parts, last = k.split(".")
        for p in parts:
            cur = cur.setdefault(p, {})
        cur[last] = v
    return out

Q: How do you do softmax numpy?
A: Same formula.
import numpy as np
def softmax(x):
    e = np.exp(x - x.max(axis=-1, keepdims=True))
    return e / e.sum(axis=-1, keepdims=True)

Q: How do you do layer norm numpy?
A: Normalize per row.
import numpy as np
def layernorm(x, g, b, eps=1e-5):
    m = x.mean(-1, keepdims=True)
    v = x.var(-1, keepdims=True)
    return g * (x - m) / np.sqrt(v + eps) + b

Q: How do you do attention numpy?
A: QK^T, scale, mask, softmax, times V.
import numpy as np
def attention(Q, K, V):
    s = Q @ K.T / np.sqrt(Q.shape[-1])
    s = np.where(np.arange(s.shape[0])[:, None] >= np.arange(s.shape[1]), s, -1e9)
    return softmax(s) @ V
